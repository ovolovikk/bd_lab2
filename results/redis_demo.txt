redis> SET lab2:card:S0000001 "{"_id": "S0000001", "last_name": "Мельник", "first_name": "Софія", "group": {"name"...
  True
redis> GET lab2:card:S0000001
  {"_id": "S0000001", "last_name": "Мельник", "first_name": "Софія", "group": {"name": "G-00001"}}
redis> TTL lab2:card:S0000001
  60
redis> EXISTS lab2:card:S9999999
  0

━━━ 2. INCR: лічильник переглядів профілю (атомарний, без гонок) ━━━
redis> INCR lab2:views:S0000001
  1
redis> INCR lab2:views:S0000001
  2
redis> INCR lab2:views:S0000001
  3
redis> INCRBY lab2:views:S0000001 10
  13

━━━ 3. HASH + EXPIRE: сесія співробітника деканату ━━━
redis> HSET lab2:session:7f3a username dekanat1 role deanery faculty_id 1
  3
redis> EXPIRE lab2:session:7f3a 1800
  True
redis> HGET lab2:session:7f3a role
  deanery
redis> HGETALL lab2:session:7f3a
  {'username': 'dekanat1', 'role': 'deanery', 'faculty_id': '1'}

━━━ 4. LIST: черга повідомлень студенту (FIFO) ━━━
redis> RPUSH lab2:inbox:S0000001 "Оцінку з БД виставлено" "Призначено стипендію" "Зміна розкладу"
  3
redis> LLEN lab2:inbox:S0000001
  3
redis> LPOP lab2:inbox:S0000001
  Оцінку з БД виставлено
redis> LRANGE lab2:inbox:S0000001 0 -1
  1) Призначено стипендію
  2) Зміна розкладу
